Konkreti klasė, paveldinti iš \doxylink{classZmogus}{Zmogus}.

\item Kopijavimo konstruktorių
\item Kopijavimo priskyrimo operatorių
\item Perkėlimo konstruktorių
\item Perkėlimo priskyrimo operatorių
\item Destruktorių (paveldi iš {\ttfamily \doxylink{classZmogus}{Zmogus}})
\end{DoxyItemize}\hypertarget{classStudentas_autotoc_md1}{}\doxysubsubsubsection{\texorpdfstring{Galutinio balo formulė}{Galutinio balo formulė}}\label{classStudentas_autotoc_md1}

  \text{galrezVid} = 0.4 \cdot \text{vidurkis} + 0.6 \cdot \text{egzaminas}
\]  \[
