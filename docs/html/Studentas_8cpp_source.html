162void Studentas::suskaiciuotiGalutini()
163{
164 std::vector<int> surikiuoti = nd;
165 std::sort(surikiuoti.begin(), surikiuoti.end());
166
167 const int n = static_cast<int>(surikiuoti.size());
168
169 if (n == 0)
170 {
171 galrezVid = 0.0f;
172 galrezMed = 0.0f;
173 return;
174 }
175
176 float med = 0.0f;
177 if (n % 2 == 1)
178 {
179 med = static_cast<float>(surikiuoti[n / 2]);
180 }
181 else
182 {
183 med = (static_cast<float>(surikiuoti[n / 2]) +
184 static_cast<float>(surikiuoti[n / 2 - 1])) / 2.0f;
185 }
186
187 galrezVid = 0.4f * vidurkis + 0.6f * static_cast<float>(egzaminas);
188 galrezMed = 0.4f * med + 0.6f * static_cast<float>(egzaminas);
189}

220std::istream& Studentas::readStudent(std::istream& is)
221{
222 std::string line;
223 if (!std::getline(is, line) || line.empty())
224 {
225 return is;
226 }
227 std::istringstream ss(line);
228
229 ss >> vardas >> pavarde;
230
231 nd.clear();
232 vidurkis = 0.0f;
233 egzaminas = 0;
234
235 int x = 0;
236
237 while (ss >> x)
238 {
239 nd.push_back(x);
240 }
241
242 if (!nd.empty())
243 {
244 egzaminas = nd.back();
245 nd.pop_back();
246
247 for (const int p : nd)
248 {
249 vidurkis += static_cast<float>(p);
250 }
251
252 if (!nd.empty())
253 {
254 vidurkis /= static_cast<float>(nd.size());
255 }
256 }
257

312void Studentas::parinktiAtsitiktinius()
313{
314 std::mt19937 gen(std::random_device{}());
315 std::uniform_int_distribution<int> dist(1, 10);
316
317 const int kiekND = dist(gen);
318
319 nd.clear();
320 vidurkis = 0.0f;
321
322 for (int i = 0; i < kiekND; i++)
323 {
324 const int balas = dist(gen);
325 nd.push_back(balas);
326 vidurkis += static_cast<float>(balas);
327 }
328
329 vidurkis /= static_cast<float>(nd.size());
330 egzaminas = dist(gen);
