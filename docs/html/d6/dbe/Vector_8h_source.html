56 //Sunaikina visus elementus
57 void destroyAll() noexcept
58 {
59 for (size_type i = 0; i < size_; ++i)
60 {
61 AllocTraits::destroy(alloc_, data_ + i);
62 }
63 }
64
65 //Atpalaiduoja paskirtą atminties bloką
66 void deallocate() noexcept
67 {
68 if (data_)
69 {
70 AllocTraits::deallocate(alloc_, data_, cap_);
71 data_ = nullptr;
72 }
73 }
74
75 //Perskirsto atmintį į naują bloką (move semantika)
76 void reallocate(size_type newCap)
77 {
78 pointer newData = AllocTraits::allocate(alloc_, newCap);
79 size_type i = 0;
80 try
81 {
82 for (; i < size_; ++i)
83 AllocTraits::construct(alloc_, newData + i,
84 std::move_if_noexcept(data_[i]));
85 }
86 catch (...)
87 {
88 for (size_type j = 0; j < i; ++j)
89 AllocTraits::destroy(alloc_, newData + j);
90 AllocTraits::deallocate(alloc_, newData, newCap);
91 throw;
92 }
93 destroyAll();
94 deallocate();
95 data_ = newData;
96 cap_ = newCap;
97 }

99 //Eksponentinis augimas (×2)
100 size_type growCap() const noexcept
101 {
102 return cap_ == 0 ? 1 : cap_ * 2;
103 }

493 /**
494 * @brief Įterpia count kopijų value prieš pos.
495 */
496 iterator insert(const_iterator pos, size_type count, const T& value)
497 {
498 const size_type idx = static_cast<size_type>(pos - data_);
499 if (count == 0)
500 {
501 return data_ + idx;
502 }
503 ensureSpace(count);
504 const size_type oldSize = size_;
505 shiftRight(idx, count);
506 for (size_type i = 0; i < count; ++i)
507 {
508 const size_type dst = idx + i;
509 if (dst < oldSize)
510 {
511 data_[dst] = value;
512 }
513 else
514 {
515 AllocTraits::construct(alloc_, data_ + dst, value);
516 }
517 }
518 size_ += count;
519 return data_ + idx;
520 }

522 /**
523 * @brief Įterpia diapazoną [first, last) prieš pos.
524 */
525 template <typename InputIt, typename = std::enable_if_t<std::is_base_of_v< std::input_iterator_tag, typename std::iterator_traits<InputIt>::iterator_category>>>
526 iterator insert(const_iterator pos, InputIt first, InputIt last)
527 {
528 const size_type idx = static_cast<size_type>(pos - data_);
529 Vector tmp(first, last, alloc_);
530 const size_type count = tmp.size_;
531 if (count == 0)
532 {
533 return data_ + idx;
534 }
535 ensureSpace(count);
536 const size_type oldSize = size_;
537 shiftRight(idx, count);
538 for (size_type i = 0; i < count; ++i)
539 {
540 const size_type dst = idx + i;
541 if (dst < oldSize)
542 {
543 data_[dst] = std::move(tmp.data_[i]);
544 }
545 else
546 {
547 AllocTraits::construct(alloc_, data_ + dst, std::move(tmp.data_[i]));
548 }
549 }
550 size_ += count;
551 return data_ + idx;
552 }

580 /**
581 * @brief Pašalina elementą pos pozicijoje.
582 * @return Iteratorius po pašalinto elemento.
583 */
584 iterator erase(const_iterator pos)
585 {
586 const size_type idx = static_cast<size_type>(pos - data_);
587 for (size_type i = idx; i + 1 < size_; ++i)
588 {
589 data_[i] = std::move(data_[i + 1]);
590 }
591 AllocTraits::destroy(alloc_, data_ + size_ - 1);
592 size_--;
593 return data_ + idx;
594 }
595
596 /**
597 * @brief Pašalina diapazoną [first, last).
598 * @return Iteratorius į pirmą elementą po pašalintų.
599 */
600 iterator erase(const_iterator first, const_iterator last)
601 {
602 const size_type idxFirst = static_cast<size_type>(first - data_);
603 const size_type idxLast = static_cast<size_type>(last - data_);
604 const size_type count = idxLast - idxFirst;
605 if (count == 0)
606 {
607 return data_ + idxFirst;
608 }
609 // Perkeliame elementus kairėn
610 for (size_type i = idxFirst; i + count < size_; ++i)
611 {
612 data_[i] = std::move(data_[i + count]);
613 }
614 // Sunaikinami galo elementai
615 for (size_type i = size_ - count; i < size_; ++i)
616 {
617 AllocTraits::destroy(alloc_, data_ + i);
618 }
619 size_ -= count;
620 return data_ + idxFirst;
621 }

742 // Vidiniai pagalbiniai metodai insert logikai

744 //Garantuoja vietos count naujiems elementams (perskirsto jei reikia)
745 void ensureSpace(size_type count)
746 {
747 if (size_ + count > cap_)
748 {
749 reallocate(std::max(size_ + count, growCap()));
750 }
751 }
752
753 /**
754 * @brief Pastumia elementus iš [idx, size_) dešinėn per count pozicijų.
755 *
756 * Elementai į nepatalpintą (uninitialized) sritį – construct,
757 * į jau egzistuojančią – move-assign.
758 * Rezultate pozicijos [idx, idx+count) – „inicializuotos, bet perkeltos"
759 * arba neinicializuotos – tvarkomos insert() metodo.
760 *
761 * @pre size_ + count <= cap_ (ensureSpace jau iškviesta)
762 */
763 void shiftRight(size_type idx, size_type count)
764 {
765 // Perkeliame dešinėn (iš dešinės į kairę, kad neverstume)
766 size_type i = size_ + count;
767 while (i-->idx + count)
768 {
769 const size_type src = i - count;
770 if (i >= size_)
771 {
772 AllocTraits::construct(alloc_, data_ + i, std::move(data_[src]));
773 }
774 else
775 {
776 data_[i] = std::move(data_[src]);
777 }
778 }
779 }

784/// @relates Vector
785template <typename T, typename A>
786bool operator==(const Vector<T, A>& l, const Vector<T, A>& r)
787{
788 if (l.size() != r.size())
789 {
790 return false;
791 }
792 for (typename Vector<T, A>::size_type i = 0; i < l.size(); ++i)
793 {
794 if (!(l[i] == r[i]))
795 {
796 return false;
797 }
798 }
799 return true;
800}
